function escapeText(s) {
  return String(s).replace(/[&<>"']/g, c => ({
    "&":"&","<":"<",">":">",'"':""","'":"'"
  }[c]));
}

  - Parses HTML with DOMParser.
  - Rejects parsererror.
  - Rejects malformed closing structure detected by a lightweight tokenizer.

  - Checks script blocks have generally plausible JavaScript using Function().
  This is deliberately not a complete HTML/CSS/JS compiler.
*/
