function Card() {
  return (
    <div className="card" onClick={{() => alert('Hei!')}}
      style={{ padding: "16px", borderRadius: "8px" }}>
      <span className="title">React</span>
    </div>
  );
}
className 

<div className="card">
  <span className="title">Hello</span>
</div>

React writes className to the HTML class attribute.
Use CSS to decide what those classes look like.
onClick 

<div onClick={{() => console.log('clicked')}}>
  <span onClick={{handleClick}}>Click me</span>
</div>

Pass a function. Do not call it while rendering:
onClick={{handleClick}}       good
onClick={{handleClick()}}      runs immediately
style 

<div style={{ padding: "12px", backgroundColor: "lightblue" }}>
  <span style={{ fontWeight: "bold", fontSize: "20px" }}>
    Hello
  </span>
</div>

style uses a JavaScript object, so CSS property names use camelCase.
background-color  →  backgroundColor

function Status() {
  const handleClick = () => console.log("status clicked");

  return (
    <div
      className="status"
      onClick={{handleClick}}
      style={{ padding: "10px", border: "1px solid gray" }}
    >
      <span
        className="status-text"
        style={{ fontWeight: "bold" }}
      >
        Ready
      </span>
    </div>
  );
}
