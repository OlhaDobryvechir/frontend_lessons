the starting value is transparent, not white
#rrggbb 

#58a6ff
 └┬┘└┬┘└┬┘
  58  a6  ff   red, green, blue — each 00 to ff, which is 0 to 255

#abc        shorthand: each digit is doubled → #aabbcc
#58a6ffcc   eight digits: the last pair is alpha → cc = 204 = 80%
rgb() 

color: rgb(88 166 255);             modern: spaces, no commas
color: rgb(88 166 255 / 50%);       alpha after a slash

color: rgb(88, 166, 255);           the older form, still valid
color: rgba(88, 166, 255, 0.5);     rgba is now just an alias for rgb
hsl() 

color: hsl(212 100% 67.25%);   exactly the same colour as #58a6ff
            │    │     └ lightness: 0% black, 100% white
            │    └ saturation: 0% grey, 100% full
            └ hue: a position on the colour wheel, 0–360

a hover state is one number away
hsl(212 100% 67.25%)   →   hsl(212 100% 75%)
